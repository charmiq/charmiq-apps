// .. the car's road: down the avenue, a swerve, the hard left onto the cross street, the ramp
const V0 = 8, START_Z = 104, SEES_Z = 28
const PTS = [[-4, START_Z], [-4, 60], [-4, 27], [-3, 21], [-0.5, 15], [2.5, 9], [5.5, 3.5], [9.5, -1.5], [14, -3.6], [20, -4], [30, -4], [49.5, -4]]
const LINE = []
for (let i = 0; i < PTS.length - 1; i++) {
  const a = PTS[Math.max(0, i - 1)], b = PTS[i], c = PTS[i + 1], d = PTS[Math.min(PTS.length - 1, i + 2)]
  for (let k = 0; k < 40; k++) {
    const u = k / 40, u2 = u * u, u3 = u2 * u
    const cr = (p0, p1, p2, p3) => 0.5 * ((2 * p1) + ((-p0 + p2) * u) + (((2 * p0) - (5 * p1) + (4 * p2) - p3) * u2) + ((-p0 + (3 * p1) - (3 * p2) + p3) * u3))
    LINE.push([cr(a[0], b[0], c[0], d[0]), cr(a[1], b[1], c[1], d[1])])
  }
}
LINE.push(PTS[PTS.length - 1])
const LEN = [0]
for (let i = 1; i < LINE.length; i++) LEN.push(LEN[i - 1] + Math.hypot(LINE[i][0] - LINE[i - 1][0], LINE[i][1] - LINE[i - 1][1]))
const TOTAL = LEN[LEN.length - 1]
const point = (s) => {
  s = Math.max(0, Math.min(TOTAL, s))
  let lo = 0, hi = LEN.length - 1
  while (hi - lo > 1) { const m = (lo + hi) >> 1; if (LEN[m] <= s) lo = m; else hi = m }
  const f = (s - LEN[lo]) / ((LEN[hi] - LEN[lo]) || 1)
  return { x: lerp(LINE[lo][0], LINE[hi][0], f), z: lerp(LINE[lo][1], LINE[hi][1], f) }
}
const heading = (s) => { const a = point(s - 0.8), b = point(s + 0.8); return Math.atan2(-(b.x - a.x), -(b.z - a.z)) * 180 / Math.PI }

// the clock along it: cruising, then his foot finds the gas after the swerve
const SEES = (START_Z - SEES_Z) / V0
const DT = 1 / 240, S = [0]
for (let t = 0, s = 0; s < TOTAL; ) { const v = Math.min(14, V0 + Math.max(0, 3 * (t - SEES - 1.2))); s += v * DT; t += DT; S.push(Math.min(s, TOTAL)) }
const sAt = (t) => S[Math.min(S.length - 1, Math.round(t / DT))]
const when = (test) => { for (let i = 0; i < S.length; i++) if (test(point(S[i]))) return i * DT; return (S.length - 1) * DT }

// the beats, in seconds, from where the car is
const PASS = when((p) => p.z <= 11)          // he passes the cat
const GATE = when((p) => p.x >= 27)          // his bumper meets the barriers
const RAMP = when((p) => p.x >= 44)          // onto the planks
const TAKEOFF = (S.length - 1) * DT          // off the top
const FLIGHT = TAKEOFF + 0.35, JOY = TAKEOFF + 1.6, WATCH = TAKEOFF + 3.6, END = TAKEOFF + 5.8
const CORNER = SEES - 2.2, HOP = SEES - 1.2
console.log('sees', SEES.toFixed(2), 'pass', PASS.toFixed(2), 'gate', GATE.toFixed(2), 'ramp', RAMP.toFixed(2), 'takeoff', TAKEOFF.toFixed(2), 'end', END.toFixed(2))

car.path(0, TAKEOFF, (u, t) => {
  const s = sAt(t), p = point(s), up = Math.max(0, Math.min(1, (p.x - 44) / 5.5))
  return { x: p.x, z: p.z, yaw: heading(s), height: 6 * up, pitch: 21.8 * Math.min(1, up / 0.15), roll: 0 }
})
// the flight: a jump's arc, the clock slowing toward the top
car.path(TAKEOFF, END, (u, t) => {
  const s = 0.95 * (1 - Math.pow(1 - u, 2.4))
  return { x: 49.5 + (20 * s), z: -4, yaw: -90, height: 6 + arc(s, 55, 58), pitch: 21.8 - (18 * u), roll: -8 * Math.sin(u * Math.PI) }
})

// .. the driver: happy at the wheel, the yank, gritted through the gate, horror, joy ......

// .. the cat: down the sidewalk to the corner, off the curb into his lane, a jump, back .....
const bob = (t) => 0.35 * Math.abs(Math.sin(t * 9))
cat.at(0, { x: -12.8, z: 28, yaw: 0 })
   .at(CORNER, { z: 11 }, 'linear')

   .at(SEES + 0.1, { yaw: -90 })
   .at(SEES + 0.4, { yaw: -180 }, 'ease-out')
   .at(PASS + 0.4, { yaw: -180, x: -4.5 })

  if (t < CORNER + 0.3) return { height: 1 }
  if (t < HOP) return { height: 1 + bob(t) }
  if (t < HOP + 0.3) { const f = (t - HOP) / 0.3; return { height: lerp(1, 0, f) + (1.2 * Math.sin(f * Math.PI)) } }
  if (t < SEES - 0.2) return { height: bob(t) }
  if (t < PASS - 0.15) return { height: 0 }
  if (t < PASS + 0.35) return { height: 2.5 * Math.sin(((t - PASS + 0.15) / 0.5) * Math.PI) }
  if (t < PASS + 0.6) return { height: 0 }
  if (t < PASS + 1.0) return { height: bob(t * 1.5) }
  if (t < PASS + 1.3) { const f = (t - PASS - 1.0) / 0.3; return { height: lerp(0, 1, f) + (1.2 * Math.sin(f * Math.PI)) } }
  return { height: 1 }
})

  const sign = (w.dir === 'north' || w.dir === 'east') ? 1 : -1, alongZ = (w.dir === 'north' || w.dir === 'south')
  a.path(0, END, (u, t) => {

    const start = alongZ ? w.z : w.x
    if (w.stop !== undefined) d = Math.min(d, Math.abs(w.stop - start))
    const walking = (w.stop === undefined || d < Math.abs(w.stop - start)) && (w.react === undefined || t < w.react)
    const swing = walking ? 28 * Math.sin(d * 1.6 + i) : 0
    const pos = alongZ ? { x: w.x, z: w.z + (sign * d) } : { x: w.x + (sign * d), z: w.z }
    const scared = (w.react !== undefined) && (t >= w.react)
    return { ...pos, height: 1, yaw: YAW[w.dir],

  })

})

## Try It

- Press play, or drag along the timeline: the film plays in the studio above.
- Press **Back to the build** in the studio to work on the set: the film waits until you press **Stage again**.
- Press **Add sound** to put a sound from your Drive into the film at the playhead, then drag it along the Sound row to move it.
- Press **Export MP4** to save the film as a video, and **Save** to keep the film itself (its set and its scene) in your Drive; **Open…** brings one back, **New** starts afresh.
- Change the scene: make the car leave the ramp faster, or the camera swing the other way. The timeline changes as you type.
- Add a second Brick Studio below this one, its build empty: it shows the scene's plan, the whole jump from above with every path and where the camera stands.
